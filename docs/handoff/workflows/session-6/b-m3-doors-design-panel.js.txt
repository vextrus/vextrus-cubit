phase('Design')
const designs = await parallel(DESIGNERS.map((d) => () =>
  agent(FACTS + '\n\n--- YOUR SLICE (' + d.key + ') ---\n' + d.prompt, {
    label: d.label,
    phase: 'Design',
    schema: DESIGN,
    effort: 'high',
  }).then((r) => (r ? Object.assign({ key: d.key }, r) : null))
))

const good = designs.filter(Boolean)
log(good.length + '/' + DESIGNERS.length + ' designers returned')

const combined = good.map((d) =>
  '### ' + d.key + '\n' + d.report +
  '\n\nDECISIONS:\n' + (d.decisions || []).map((x) => '- Q: ' + x.question + '\n  RULING: ' + x.ruling + '\n  BECAUSE: ' + x.because).join('\n') +
  '\n\nFILES:\n' + (d.files || []).map((f) => '- ' + f.path + ' [' + (f.lane || '?') + '] — ' + f.change).join('\n') +
  '\n\nRISKS:\n' + (d.risks || []).join('\n')
).join('\n\n---\n\n')

phase('Synthesis')
const plan = await agent(
  FACTS +
  '\n\n--- SIX DESIGNS ---\n' + combined +
  '\n\n--- YOUR JOB ---\nYou are the integrator. Produce ONE increment plan an orchestrator can brief implementers from tonight.\n\n' +
  'Required of you:\n' +
  '(a) RESOLVE every disagreement between the six designs explicitly. Where two designers propose different shapes for the same thing, pick one and say why the other is worse. Do not paper over a conflict.\n' +
  '(b) Produce THE FILE PARTITION: a numbered list of implementer slices such that NO TWO SLICES SHARE A FILE. That is a hard constraint. Both doors touch rebuild.ts, PlacementRow, detect.ts, the placements schema, migration 0056, schema-aggregate.test.ts, the spec, the fixme roster and the notation corpus test — so slice by LAYER, not by door, and name which slice owns each shared file. Give each slice a dependency order and say which must be serial.\n' +
  '(c) Name the ONE migration 0056 and list every column it carries, from all slices.\n' +
  '(d) Give the exact Interpretation numbers and the document each lands in.\n' +
  '(e) State the ORDER OF PROOF: which lane proves which slice, and the exact commands.\n' +
  '(f) Name the three things most likely to go wrong, with the early check that would catch each.\n' +
  '(g) If you believe any part of this increment cannot be done as specified, say so plainly with the reason — that is more valuable than a plan that fails at hour three.',
  { label: 'synthesise:increment-plan', phase: 'Synthesis', schema: DESIGN, effort: 'max' }
)

return { designs: good, plan }
