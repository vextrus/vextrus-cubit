phase('Read')
const reports = await parallel(READERS.map((r) => () =>
  agent(BOUNDARY + '\n\n--- YOUR SLICE (' + r.key + ') ---\n' + r.prompt, {
    label: r.label,
    phase: 'Read',
    schema: READER,
  }).then((res) => (res ? Object.assign({ key: r.key }, res) : null))
))

const good = reports.filter(Boolean)
log(good.length + '/' + READERS.length + ' readers returned')

phase('Critic')
const combined = good.map((r) =>
  '### ' + r.key + '\n' + r.report +
  '\n\nKEY FACTS:\n' + (r.keyFacts || []).map((k) => '- ' + k.fact + ' [' + k.location + ']').join('\n') +
  '\n\nRISKS:\n' + (r.risks || []).join('\n')
).join('\n\n---\n\n')

const critic = await agent(
  BOUNDARY +
  '\n\nBelow is a combined orientation map assembled by eleven readers of the vextrus-cubit repo, for an orchestrator about to brief implementers on two doors (a CIRCULAR column section read off %%C450, and a plan member OWN storey range read off its note STARTS AT 1F), four unwired Jev passes, and the shell effect-handover slots.\n\n' +
  'Your job is the completeness critic: name what is MISSING or UNVERIFIED that a brief would need. Specifically check, by reading the repo yourself where you doubt a claim: (a) any claim above that is wrong or unsupported — verify the three you most doubt and say what you found; (b) any file that two of these workstreams would BOTH have to edit (two briefs must never share a file) — list the collisions explicitly; (c) the next free Interpretation number; (d) anything about the golden band arithmetic that is still a guess.\n\n' +
  '--- COMBINED MAP ---\n' + combined,
  { label: 'critic:completeness', phase: 'Critic', schema: READER, effort: 'high' }
)

return { readers: good, critic }
