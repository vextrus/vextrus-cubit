export const meta = {
  name: 's8-wave',
  description: 'Session 8 implementation wave: each slice implemented in its own worktree, adversarially reviewed, then fixed; read-only refuters alongside',
  phases: [
    { title: 'Implement', detail: 'one implementer per slice, in its own git worktree, committing on its branch' },
    { title: 'Review', detail: 'a refuter reads each slice diff against its spec and the law' },
    { title: 'Fix', detail: 'the implementer role returns to its worktree to address a FIX_REQUIRED verdict' },
    { title: 'Refute', detail: 'read-only verification agents' },
  ],
}

const MAIN = '/home/riz/vextrus-cubit'
const W = args.wave
const BASE = args.base

const RESULT = {
  type: 'object',
  properties: {
    slice: { type: 'string' },
    worktree: { type: 'string', description: 'absolute path of your worktree (pwd)' },
    branch: { type: 'string' },
    commits: { type: 'array', items: { type: 'string' }, description: 'full shas, oldest first, on your branch after BASE' },
    status: { type: 'string', enum: ['DONE', 'PARTIAL', 'BLOCKED'] },
    summary: { type: 'string', description: 'what a QS can now do or see, and what changed, in plain words' },
    lanes: { type: 'array', items: { type: 'object', properties: { command: { type: 'string' }, verdict: { type: 'string', description: 'the lane own verdict line, quoted' } }, required: ['command', 'verdict'] } },
    unrun: { type: 'array', items: { type: 'string' }, description: 'proofs you could not run here (db lane refused, e2e, browser) and exactly what the integrator must run' },
    lawPlaceholders: { type: 'array', items: { type: 'object', properties: { token: { type: 'string' }, where: { type: 'string' }, gist: { type: 'string' } }, required: ['token', 'where', 'gist'] } },
    decisionsAmended: { type: 'array', items: { type: 'string' } },
    sharedFilesTouched: { type: 'array', items: { type: 'string' }, description: 'registries/strings/testids/errors/journeys you touched (merge hot spots)' },
    picturesLikelyMoved: { type: 'array', items: { type: 'string' } },
    readbackChange: { type: 'string', description: 'which stored figures should move on the next J-000 read-back, and why; "none" if none' },
    risks: { type: 'array', items: { type: 'string' } },
  },
  required: ['slice', 'worktree', 'branch', 'commits', 'status', 'summary', 'lanes', 'unrun', 'lawPlaceholders', 'readbackChange'],
}

const VERDICT = {
  type: 'object',
  properties: {
    verdict: { type: 'string', enum: ['PASS', 'FIX_REQUIRED'] },
    blocking: { type: 'array', items: { type: 'object', properties: { what: { type: 'string' }, where: { type: 'string' }, why: { type: 'string' }, fix: { type: 'string' } }, required: ['what', 'where', 'why', 'fix'] } },
    advisory: { type: 'array', items: { type: 'string' } },
    checked: { type: 'string', description: 'what you actually verified and how' },
  },
  required: ['verdict', 'blocking', 'checked'],
}

function implementPrompt(s) {
  const tag = s.id.replace(/[^A-Za-z0-9]/g, '')
  return `You are implementing slice ${s.id} of Vextrus Cubit's session 8, wave ${W}, in YOUR OWN git worktree (your current directory; check with pwd and git status). The main checkout is ${MAIN}: NEVER write there. Your branch starts at ${BASE}.

READ FIRST (absolute paths into the main checkout):
- your slice spec: ${MAIN}/.private/work/session-8/slices/${s.specFile || s.id}.json (scope, files, dependsOn, proof, qsOutcome, and the wave's shared-file owners);
${(s.maps || []).map((m) => `- the area map and its adversarial critic's corrections: ${MAIN}/.private/work/session-8/maps/by-area/${m}.md (apply the critic's corrections that concern your slice)`).join('\n')}
${(s.extra || []).map((e) => `- ${e}`).join('\n')}
- the path-scoped law for what you touch (${MAIN}/.claude/rules/*.md) and the Design Decision(s) you amend (docs/design/<screen>.md).
${s.note ? `\nORCHESTRATOR'S NOTE: ${s.note}\n` : ''}
HOW TO WORK:
- Build the slice to its spec, vertically: the QS outcome it names, in the running product's code paths, never a stub. Where the spec and the code disagree, the code at HEAD and the Bible win; say so in risks.
- Stay inside your slice's files. Shared registries (src/ui/testids.ts, src/ui/strings/*, src/core/errors/*, fixme roster, aggregate digests) may be touched where your code needs them: add in a contiguous block, and list them in sharedFilesTouched — the integrator merges them serially.
- LAW IDS: never allocate a real number. Write each new Interpretation as the token I-${tag}-a, I-${tag}-b, ... and each Deviation as D-${tag}-a, ... (in docs/design/*.md, docs/decisions/deviations.md, code comments and tests alike); the integrator renumbers them in order. Report every token you used. Never write prose ABOUT the tokens (no 'tokens the integrator renumbers' sentences) — write the Decision as if the ids were real. MIGRATIONS: if your slice needs one, generate it the tree's normal way as the next number your worktree sees (drizzle-kit generate from src/core/db/schema-<area>.ts, --name <kebab-name>), and put every hand-written statement (RLS policies, grants, triggers, data INSERTs such as ruleset editions) in ONE block at the END of the .sql file, opened by the line '-- hand-written (session 8, ${s.id}): <what>' — parallel slices will collide on the number and the snapshot, and the integrator regenerates the generated part in merge order and re-appends your block. Report the migration in sharedFilesTouched.
- A Design Decision you change is amended in the same commit (C-13). Every user-facing string goes in the string tables; every number through the format seam.
- TESTS WITH TEETH: every behaviour you add or fix has a test that is red without your change (check it: stash or revert the line and see it fail, then restore). Run the narrow lanes and quote their verdict lines: pnpm vitest run <your test files>; npx tsc --noEmit -p . (the types lane); npx eslint <your changed files>; for cad: uv run --project cad pytest <tests> -q and uv run --project cad ruff check cad. The db lane (pnpm test:db <files>) refuses while a served product stands (a demo may be up): if it refuses, write the db tests anyway and list them in unrun. NEVER run pnpm e2e, pnpm verify, pnpm gate or pnpm demo here (worktrees cannot build the product); list the journeys and pictures the integrator must run/look at in unrun and picturesLikelyMoved.
- COMMIT IN STAGES as you go (a WIP commit per coherent step is fine — a power cut on this machine has already cost a wave its uncommitted work twice; the integrator picks the whole branch).
- COMMIT on your branch with explicit paths (git add <paths>; the guard refuses git add -A and commit -a), subjects in the tree's style ("<area>: <what changed and why>"). A regenerated fixture or corpus goes in its own "baseline: ..." commit naming the proof. Leave no uncommitted changes.
- Never loosen a proof, never test.skip, never fake a leg. "Measure less, completely, and say so."
Return the result (schema).`
}

function reviewPrompt(s, r) {
  return `Adversarially review slice ${s.id} of Vextrus Cubit session 8 (wave ${W}). Its spec: ${MAIN}/.private/work/session-8/slices/${s.specFile || s.id}.json${(s.maps || []).length ? `; its area map + critic: ${(s.maps || []).map((m) => `${MAIN}/.private/work/session-8/maps/by-area/${m}.md`).join(', ')}` : ''}.
The implementer reports (JSON): ${JSON.stringify(r).slice(0, 12000)}
Its commits are in the shared object store: read them from the main checkout with git show / git diff ${BASE}..${(r.commits || []).slice(-1)[0] || '<head>'} (and read files at that commit with git show <sha>:<path>); its worktree is ${r.worktree} (READ-ONLY for you — you may run tests there: pnpm vitest run <files>, npx tsc --noEmit -p ., cad pytest; never edit, never commit).
Try to prove the slice WRONG where it matters: (1) does it deliver the spec's QS outcome in the real code path (not a stub, not only tests)? (2) does it break the law (over-measurement, an undeclared omission, a number not through the format seam, a string not in the tables, an act that bypasses the act seam, a writer of quantity lines other than the gate, an import against ARCH-01, a second home for an invariant)? (3) do its tests have teeth (would they fail without the change)? re-run them; (4) what will go red at integration that the implementer could not run (e2e journeys, db suites, pictures, the aggregate digests, the testid ratchet, the fixme roster)? name them; (5) scope creep into files another slice owns (the spec's sharedOwners). Verdict PASS only if nothing blocking remains; list blocking items with a concrete fix each.`
}

function fixPrompt(s, r, v) {
  return `You are returning to slice ${s.id} (Vextrus Cubit session 8, wave ${W}) to address an adversarial review. Work ONLY in the worktree ${r.worktree} (cd there first; confirm with git status and git log). Never write to ${MAIN}.
The slice spec: ${MAIN}/.private/work/session-8/slices/${s.specFile || s.id}.json. The implementer's report: ${JSON.stringify(r).slice(0, 8000)}
The review's BLOCKING items (fix each, or refute it with evidence in risks if it is wrong): ${JSON.stringify(v.blocking)}
Advisory (take what is cheap and right): ${JSON.stringify(v.advisory || [])}
Same rules as the implementer: law ids as I-/D- placeholder tokens (I-${s.id.replace(/[^A-Za-z0-9]/g, '')}-x), tests with teeth, narrow lanes run and quoted, no e2e/verify/gate/demo, commit with explicit paths, nothing left uncommitted. Return the UPDATED full result (schema) with all commits since ${BASE}.`
}

const slices = args.slices || []
const refutes = args.refutes || []

const refuteRuns = parallel(refutes.map((q) => () => agent(q.prompt, { label: `refute:${q.key}`, phase: 'Refute', agentType: 'refuter', schema: q.schema || { type: 'object', properties: { verdict: { type: 'string' }, findings: { type: 'array', items: { type: 'object', properties: { claim: { type: 'string' }, verdict: { type: 'string', enum: ['CONFIRMED', 'REFUTED', 'UNCLEAR'] }, evidence: { type: 'string' }, consequence: { type: 'string' } }, required: ['claim', 'verdict', 'evidence'] } }, report: { type: 'string' } }, required: ['verdict', 'findings', 'report'] } }).then((x) => ({ key: q.key, result: x }))))

const built = await pipeline(
  slices,
  (s) => agent(implementPrompt(s), { label: `impl:${s.id}`, phase: 'Implement', isolation: 'worktree', schema: RESULT }),
  (r, s) => (r && r.commits && r.commits.length ? agent(reviewPrompt(s, r), { label: `review:${s.id}`, phase: 'Review', agentType: 'refuter', schema: VERDICT }).then((v) => ({ r, v })) : { r, v: null }),
  (rv, s) => {
    if (!rv || !rv.r) return { slice: s.id, result: null, review: null, fixed: null }
    if (!rv.v || rv.v.verdict === 'PASS') return { slice: s.id, result: rv.r, review: rv.v, fixed: null }
    return agent(fixPrompt(s, rv.r, rv.v), { label: `fix:${s.id}`, phase: 'Fix', schema: RESULT }).then((f) => ({ slice: s.id, result: rv.r, review: rv.v, fixed: f }))
  },
)
const refuted = await refuteRuns
return { wave: W, base: BASE, built, refuted }
