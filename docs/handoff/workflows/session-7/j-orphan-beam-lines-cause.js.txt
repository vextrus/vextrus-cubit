  type: 'object',
  properties: {
    cause: { type: 'string', description: 'the named cause, one paragraph, with file:line' },
    mechanism: { type: 'array', items: { type: 'string' }, description: 'the step-by-step sequence that leaves the orphans' },

phase('Refute')
const REFUTE = {
  type: 'object',
  properties: {
    refuted: { type: 'boolean' },
    broken_steps: { type: 'array', items: { type: 'string' }, description: 'mechanism steps or citations that the code contradicts' },
    corrected: { type: 'string', description: 'the cause as the code actually supports it' },
  },
  required: ['refuted', 'broken_steps', 'corrected'],
}
const lenses = [
  'Open every file:line the synthesis cites and check each mechanism step against the code as written. A step the code does not do is broken.',
  'Try an ALTERNATIVE explanation for the 50 orphans (e.g. the lines are intentionally kept as history, a later publish is supposed to supersede by view, the register join is the design, or the orphans come from a different path than claimed) and test it against the code. If the alternative fits better, refuted=true.',
]
const votes = await parallel(lenses.map((lens, i) => () =>
  agent(`${EVIDENCE}\n\nThe synthesized cause:\n${JSON.stringify(synth, null, 1)}\n\nYOUR LENS: ${lens}\nREAD-ONLY. Default to refuted=false only if you verified the cited steps yourself.`,
    { label: `refute:${i}`, phase: 'Refute', schema: REFUTE })))

return { synth, votes: votes.filter(Boolean), investigators: got.map((g) => ({ angle: g.angle, summary: g.summary, open: g.open })) }
