// @vitest-environment jsdom
/**
 * THE FRAME'S TOOL ROW, READOUT AND INSPECTOR ARE DRAWN IN PLACE, NOT HANDED OVER (I-312; R-UI-080,
 * Design Direction 00 §1, §3.1).
 *
 * Each of the three regions was one value in the frame: a screen's effect set its node on mount and
 * set null on cleanup. Hydration stands a screen twice for about a hundred milliseconds (CLAUDE.md's
 * standing fact), and with two mounts of one screen standing, the FIRST one leaving emptied all three
 * while the second still stood — the Measure door's mechanism (session 5, `ef3bb0b5`) on the frame's
 * own regions. A screen now draws its node into the element the frame renders for the region, through
 * a portal, in the same commit as the rest of the screen, and it is gone the moment the screen is.
 *
 * The law these mounts are held to is the frame's, read as a customer meets it: the frame holds ONE
 * tool row, ONE readout and ONE inspector with ONE subject however many mounts stand; a region nothing
 * is drawn into is absent (the track collapses, the frame's own readout stands, no inspector at all);
 * and a screen mounted with no frame keeps its regions where it renders them.
 */
import { act, cleanup, render, screen } from "@testing-library/react";
import { useMemo, useState } from "react";
import { afterEach, describe, expect, test } from "vitest";
import { AppShell } from "@/ui/shell/app-shell";
import { ShellToolbar, StatusBar, useInspector, useShellStatus, useShellToolbar } from "@/ui/shell";
import { strings } from "@/ui/strings";
import { TESTIDS } from "@/ui/testids";

afterEach(cleanup);

const WORKSPACE = { tenantId: "A", name: "Acme Holdings" };
const noop = (): void => undefined;

/**
 * One screen that fills all three regions, and renders what each hook answers — the one spelling of
 * the contract every screen in the product uses.
 */
function Screen({ says, selected = true }: { says: string; selected?: boolean }) {
  const tools = useMemo(
    () => (
      <ShellToolbar label={`${says} tools`}>
        <span>{`${says} tools`}</span>
      </ShellToolbar>
    ),
    [says],
  );
  const readout = useMemo(() => <StatusBar cells={{ sheet: says }} />, [says]);
  const detail = useMemo(() => (selected ? <p>{`${says} detail`}</p> : null), [says, selected]);
  const toolbar = useShellToolbar(tools);
  const status = useShellStatus(readout);
  const inspector = useInspector(detail);
  return (
    <>
      {toolbar}
      <p>{`${says} body`}</p>
      {status}
      {inspector}
    </>
  );
}

/** The frame, with two mounts of one screen that can each be taken down on its own. */
function Stage({ first, second }: { first: boolean; second: boolean }) {
  return (
    <AppShell workspace={WORKSPACE} area="projects" atAreaHome={false} email={null} signOut={noop}>
      {first ? <Screen says="first" /> : null}
      {second ? <Screen says="second" /> : null}
    </AppShell>
  );
}

/** What the frame's tool track holds, as text. */
function toolRow(): string {
  return screen.getByTestId(TESTIDS.shell.toolbarSlot).textContent?.trim() ?? "";
}

/** Whether the body keeps the tool track open (`[data-toolbar]`, the grid's own switch). */
function trackOpen(): string | null {
  return document.querySelector(".cx-shell-body")?.getAttribute("data-toolbar") ?? null;
}

/** Every readout standing in the document, and the sheet cell each one states. */
function readoutSheets(): string[] {
  return screen.queryAllByTestId(TESTIDS.shell.status).map((line) => line.querySelector('[data-cell="sheet"] .cx-shell-status-cell-value')?.textContent ?? "");
}

/** Every inspector standing in the document, and the text of each one's body. */
function inspectorBodies(): string[] {
  return screen.queryAllByTestId(TESTIDS.shell.inspector).map((aside) => aside.querySelector(".cx-shell-inspector-body")?.textContent?.trim() ?? "");
}

/** The three regions as a customer meets them, read at once so a failure names every one it lost. */
function regions(): { tools: string; track: string | null; readouts: string[]; inspectors: string[] } {
  return { tools: toolRow(), track: trackOpen(), readouts: readoutSheets(), inspectors: inspectorBodies() };
}

describe("I-312: the frame's regions are drawn in place by the screen standing in them", () => {
  test("two mounts of one screen stand and the first leaves: the second's tool row, readout and inspector still stand (the hydration window)", () => {
    const view = render(<Stage first={true} second={true} />);
    view.rerender(<Stage first={false} second={true} />);
    expect(regions(), "what the standing mount draws is still in the frame: its tool row in an open track, the one readout, the one inspector").toEqual({
      tools: "second tools",
      track: "true",
      readouts: ["second"],
      inspectors: ["second detail"],
    });
  });

  test("the newest mount leaving leaves the first's regions standing", () => {
    const view = render(<Stage first={true} second={true} />);
    view.rerender(<Stage first={true} second={false} />);
    expect(regions()).toEqual({ tools: "first tools", track: "true", readouts: ["first"], inspectors: ["first detail"] });
  });

  test("while two mounts stand, the frame holds ONE tool row, ONE readout and ONE inspector with ONE subject (R-UI-080)", () => {
    render(<Stage first={true} second={true} />);
    expect(screen.getAllByTestId(TESTIDS.shell.toolbar), "one tool row in the 32 px track").toHaveLength(1);
    expect(readoutSheets(), "one readout in the 24 px track").toHaveLength(1);
    const bodies = inspectorBodies();
    expect(bodies, "one right column — no screen renders a second one, ever").toHaveLength(1);
    expect(["first detail", "second detail"], "and it states one subject, whole").toContain(bodies[0]);
  });

  test("leaving the screen takes all three away: the track collapses, the frame's own readout stands, no inspector at all", () => {
    const view = render(<Stage first={true} second={false} />);
    view.rerender(<Stage first={false} second={false} />);
    expect(toolRow(), "nothing is drawn to state an absence").toBe("");
    expect(trackOpen(), "the track is zero high rather than an empty strip").toBe("false");
    expect(readoutSheets(), "the frame's own readout stands, stating no sheet — not the one the screen that left named").toEqual([strings.shell_status_absent]);
    expect(inspectorBodies(), "absent — width 0, not a placeholder").toEqual([]);
  });

  test("with nothing selected there is no inspector; a selection draws it, and clearing it takes it away", () => {
    function Selecting() {
      const [selected, setSelected] = useState(false);
      return (
        <>
          <Screen says="only" selected={selected} />
          <button type="button" onClick={() => setSelected((held) => !held)}>
            toggle
          </button>
        </>
      );
    }
    render(
      <AppShell workspace={WORKSPACE} area="projects" atAreaHome={false} email={null} signOut={noop}>
        <Selecting />
      </AppShell>,
    );
    const toggle = screen.getByRole("button", { name: "toggle" });
    expect(inspectorBodies(), "nothing selected, no column").toEqual([]);
    act(() => toggle.click());
    expect(inspectorBodies()).toEqual(["only detail"]);
    act(() => toggle.click());
    expect(inspectorBodies(), "the selection cleared, the column is gone").toEqual([]);
    expect(toolRow(), "and the tool row the screen still draws is untouched").toBe("only tools");
  });

  test("a region whose node changes is redrawn in place, never doubled", () => {
    function Changing() {
      const [n, setN] = useState(1);
      const tools = useMemo(
        () => (
          <ShellToolbar label="changing">
            <span>{`press ${n}`}</span>
          </ShellToolbar>
        ),
        [n],
      );
      const drawn = useShellToolbar(tools);
      return (
        <>
          {drawn}
          <button type="button" onClick={() => setN((held) => held + 1)}>
            bump
          </button>
        </>
      );
    }
    render(
      <AppShell workspace={WORKSPACE} area="projects" atAreaHome={false} email={null} signOut={noop}>
        <Changing />
      </AppShell>,
    );
    expect(toolRow()).toBe("press 1");
    act(() => screen.getByRole("button", { name: "bump" }).click());
    expect(toolRow()).toBe("press 2");
    expect(screen.getAllByTestId(TESTIDS.shell.toolbar)).toHaveLength(1);
  });

  test("a screen mounted with no frame keeps its tool row, readout and inspector where it renders them", () => {
    render(<Screen says="alone" />);
    expect(screen.getByTestId(TESTIDS.shell.toolbar).textContent).toBe("alone tools");
    expect(readoutSheets()).toEqual(["alone"]);
    expect(screen.getByText("alone detail")).toBeTruthy();
  });
});
