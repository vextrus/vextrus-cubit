"use client";
/**
 * HOW A SCREEN FILLS A REGION OF THE FRAME — the one way, for the tool row, the readout and the
 * inspector alike (I-312; Direction §1, §3.1; R-UI-080).
 *
 * A screen DRAWS its node into the region, in place: the frame renders the region's element (the
 * HOST) and the screen renders `createPortal(node, host)` as part of its own tree, so what it draws
 * stands in the same commit as the screen and is gone the moment the screen is. Nothing is HANDED
 * OVER, and the reason is a rule of this tree: a screen can stand twice — hydration stands the
 * register twice for about a hundred milliseconds — and nothing may depend on the order its two
 * mounts leave in. A region held as ONE value, set by a screen's effect on mount and cleared on its
 * cleanup, is emptied by the first of two standing mounts leaving while the second still stands; that
 * is the mechanism that takes the register's Measure door on a customer's second visit (the lane's
 * tabs row, `ef3bb0b5`), and it takes a tool row, a readout and an inspector the same way.
 *
 * WHAT THE FRAME KNOWS, AND WHY IT IS NOT A HANDOVER. The law is ONE of each region — one tool
 * row, one readout, one inspector with one subject (R-UI-080: "no screen renders a second right
 * column, ever") — so with two mounts standing exactly one of them draws. The frame therefore holds
 * which mounts are STANDING with something to draw: each mount enters under a token of its own and
 * leaves under the same token, so a mount leaving can only ever take itself away. The frame never
 * holds a node. The newest standing mount draws; when it leaves, the next one standing draws, in the
 * same commit; the region is empty only when no standing mount has anything to draw. Which of two
 * mounts of one screen draws is the only thing the order can decide, and they draw the same thing —
 * WHETHER a region stands never depends on the order at all.
 *
 * The frame also reads that roll for the one thing only it can decide: whether the region stands at
 * all — the tool track opens, the frame's own readout stands aside, the right column exists (R-UI-080:
 * "absent — width 0, not a placeholder sentence — when nothing is selected").
 *
 * A mount enters in a LAYOUT effect, and the frame's host arrives by the element's own ref, so the
 * region, its host and the node drawn into it arrive in the same synchronous pass before the browser
 * paints, never one frame after the screen that owns them.
 */
import { useCallback, useLayoutEffect, useMemo, useState, type ReactNode } from "react";
import { createPortal } from "react-dom";

/** One region of the frame, as the frame publishes it to the screens inside it. */
export interface DrawnRegion {
  /** The element the frame renders for the region, once it is in the document. */
  readonly host: HTMLElement | null;
  /** The standing mount that draws the region, or null when no standing mount has anything to draw. */
  readonly drawer: symbol | null;
  /** The ref the frame puts on the region's element. */
  readonly setHost: (element: HTMLElement | null) => void;
  /** A mount with something to draw enters under its own token; the answer is how it leaves. */
  readonly stand: (mount: symbol) => () => void;
}

/** The frame's side of one region: the host it renders, and the roll of mounts standing in it. */
export function useDrawnRegion(): DrawnRegion {
  const [host, setHost] = useState<HTMLElement | null>(null);
  const [standing, setStanding] = useState<readonly symbol[]>([]);
  const stand = useCallback((mount: symbol): (() => void) => {
    setStanding((held) => (held.includes(mount) ? held : [...held, mount]));
    return () => setStanding((held) => held.filter((token) => token !== mount));
  }, []);
  const drawer = standing.at(-1) ?? null;
  return useMemo<DrawnRegion>(() => ({ host, drawer, setHost, stand }), [host, drawer, stand]);
}

/** Whether a node is something to draw: nothing, `null`, `undefined` and `false` are absences. */
function drawable(node: ReactNode): boolean {
  return node !== null && node !== undefined && node !== false;
}

/**
 * The screen's side: hand it the region and the node, and RENDER what comes back. Inside a frame it
 * answers the node drawn into the region's host when this mount is the one that draws, and null
 * otherwise (the host not yet in the document, or another standing mount drawing). With no frame it
 * answers the node itself, so a screen mounted alone — a jsdom suite, the gallery's evidence
 * renderer — keeps what it would have drawn where it stands, and no component is ever made to know
 * whether it is inside the shell.
 */
export function useDrawInto(region: DrawnRegion | null, node: ReactNode): ReactNode {
  // A token of this mount's own, never shared: two mounts of one screen are two tokens, so neither
  // can take the other's place away when it leaves.
  const [mount] = useState(() => Symbol("frame-region-mount"));
  const drawing = drawable(node);
  const stand = region?.stand;
  useLayoutEffect(() => (stand === undefined || !drawing ? undefined : stand(mount)), [stand, mount, drawing]);
  if (region === null) return drawing ? node : null;
  if (!drawing || region.drawer !== mount || region.host === null) return null;
  return createPortal(node, region.host);
}
